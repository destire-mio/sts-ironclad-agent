cmake_minimum_required(VERSION 3.19)
project(sts_parity_audit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SIM_ROOT "" CACHE PATH "Copied and hash-verified simulator sources")
set(JSON_INCLUDE "" CACHE PATH "Directory containing nlohmann/json.hpp")
file(GLOB_RECURSE CORE_SOURCES CONFIGURE_DEPENDS "${SIM_ROOT}/src/*.cpp")
add_library(sts_core STATIC ${CORE_SOURCES})
target_include_directories(sts_core PUBLIC "${SIM_ROOT}/include" "${SIM_ROOT}/bindings" "${JSON_INCLUDE}")
target_compile_options(sts_core PRIVATE -O2 -UNDEBUG)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(slaythespire NO_EXTRAS "${SIM_ROOT}/bindings/slaythespire.cpp" "${SIM_ROOT}/bindings/bindings-util.cpp")
target_compile_options(slaythespire PRIVATE -O2 -UNDEBUG)
target_link_libraries(slaythespire PRIVATE sts_core)
add_executable(outside_probe ../alignment/tests/outside_probe.cpp)
target_link_libraries(outside_probe PRIVATE sts_core)
add_executable(colosseum_rng_probe ../alignment/tests/colosseum_rng_probe.cpp)
target_link_libraries(colosseum_rng_probe PRIVATE sts_core)
add_executable(relic_lifecycle_probe ../alignment/tests/relic_lifecycle_probe.cpp)
target_link_libraries(relic_lifecycle_probe PRIVATE sts_core)
add_executable(relic_acquire_probe ../alignment/tests/relic_acquire_probe.cpp)
target_link_libraries(relic_acquire_probe PRIVATE sts_core)
add_executable(event_entry_probe ../alignment/tests/event_entry_probe.cpp)
target_link_libraries(event_entry_probe PRIVATE sts_core)
add_executable(event_rewards_probe ../alignment/tests/event_rewards_probe.cpp)
target_link_libraries(event_rewards_probe PRIVATE sts_core)
add_executable(treasure_probe ../alignment/tests/treasure_probe.cpp)
target_link_libraries(treasure_probe PRIVATE sts_core)
add_executable(shop_continuation_probe ../alignment/tests/shop_continuation_probe.cpp)
target_link_libraries(shop_continuation_probe PRIVATE sts_core)
add_executable(save_entry_probe ../alignment/tests/save_entry_probe.cpp)
